function nextPow2(n) { return Math.pow(2, Math.ceil(Math.log2(n))); }

07 字母頻率統計

const freq = {};
for (const ch of str) freq[ch] = (freq[ch]||0) + 1;

08 陣列反轉 手動演算法

for (let i = 0; i < n/2; i++) {
[arr[i], arr[n-1-i]] = [arr[n-1-i], arr[i]];
}
請輸入陣列後點擊反轉

09 尋找最高分

students.reduce((max, s) => s.score > max.score ? s : max)
